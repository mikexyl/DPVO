cmake_minimum_required(VERSION 3.18)
project(dpvo_teaser LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Eigen3 REQUIRED NO_MODULE PATHS /usr/share/eigen3/cmake NO_DEFAULT_PATH)
find_package(OpenMP REQUIRED)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
set(PYBIND11_FINDPYTHON ON)
add_subdirectory(${PYBIND11_SOURCE} pybind11 EXCLUDE_FROM_ALL)
set(PMC_BUILD_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory(${PMC_SOURCE} pmc EXCLUDE_FROM_ALL)
# Compile the upstream registration implementation without the unrelated PLY,
# FPFH and certification targets. No upstream solver source is modified.
add_library(teaser_registration STATIC
  ${TEASER_SOURCE}/teaser/src/registration.cc
  ${TEASER_SOURCE}/teaser/src/graph.cc)
target_include_directories(teaser_registration PUBLIC ${TEASER_SOURCE}/teaser/include)
target_link_libraries(teaser_registration PUBLIC Eigen3::Eigen OpenMP::OpenMP_CXX PRIVATE pmc)
pybind11_add_module(_teaser NO_EXTRAS bindings.cpp)
target_link_libraries(_teaser PRIVATE teaser_registration)
set_target_properties(_teaser PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../dpvo")
add_executable(teaser_audit audit.cpp)
target_link_libraries(teaser_audit PRIVATE teaser_registration)
option(TEASER_SANITIZE "Instrument native registration audit with ASan/UBSan" OFF)
if(TEASER_SANITIZE)
  foreach(target pmc teaser_registration teaser_audit)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
