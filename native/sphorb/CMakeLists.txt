cmake_minimum_required(VERSION 3.18)
project(dpvo_sphorb LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenCV 4.5.4 EXACT REQUIRED COMPONENTS core imgproc imgcodecs features2d
             PATHS /usr/lib/x86_64-linux-gnu/cmake/opencv4 NO_DEFAULT_PATH)
set(OPENSSL_CRYPTO_LIBRARY /usr/lib/x86_64-linux-gnu/libcrypto.so CACHE FILEPATH "System libcrypto" FORCE)
set(OPENSSL_INCLUDE_DIR /usr/include CACHE PATH "System OpenSSL headers" FORCE)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
set(PYBIND11_FINDPYTHON ON)
add_subdirectory(${PYBIND11_SOURCE} pybind11 EXCLUDE_FROM_ALL)
add_library(sphorb_core STATIC core.cpp detector.cpp nonmax.cpp virtual_scene.cpp)
target_include_directories(sphorb_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(sphorb_core PUBLIC ${OpenCV_LIBS} OpenSSL::Crypto Threads::Threads)
pybind11_add_module(_sphorb NO_EXTRAS bindings.cpp virtual_bindings.cpp)
target_link_libraries(_sphorb PRIVATE sphorb_core)
set_target_properties(_sphorb PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../dpvo")
option(SPHORB_SANITIZE "Build native ASan/UBSan audit" OFF)
if(SPHORB_SANITIZE)
  target_compile_options(sphorb_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sphorb_core PUBLIC -fsanitize=address,undefined)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/audit.cpp")
  add_executable(sphorb_audit audit.cpp reference/reference.cpp reference/pfm.cpp)
  target_link_libraries(sphorb_audit PRIVATE sphorb_core)
endif()
